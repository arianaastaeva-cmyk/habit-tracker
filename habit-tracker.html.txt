<div id="habitList"></div>
<script>
let habits = JSON.parse(localStorage.getItem("habits")) || [];
function saveHabits() {
localStorage.setItem("habits", JSON.stringify(habits));
}
function render() {
const list = document.getElementById("habitList");
list.innerHTML = "";
habits.forEach((habit, index) => {
const today = new Date().toISOString().split("T")[0];
const completedToday = habit.dates.includes(today);
const div = document.createElement("div");
div.className = "habit";
div.innerHTML = `
<div>
<strong>${habit.name}</strong>
<div class="counter">Выполнено: ${habit.dates.length} дней</div>
</div>
<button onclick="toggleHabit(${index})">
${completedToday ? "✔" : "Отметить"}
</button>
`;
list.appendChild(div);
});
}
function addHabit() {
const input = document.getElementById("newHabit");
const name = input.value.trim();
if (!name) return;
habits.push({ name: name, dates: [] });
input.value = "";
saveHabits();
render();
}
function toggleHabit(index) {
const today = new Date().toISOString().split("T")[0];
const habit = habits[index];
if (!habit.dates.includes(today)) {
habit.dates.push(today);
}
saveHabits();
render();
}
render();
</script>
